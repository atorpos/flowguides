function ChangeLang(){
	//
	var encodeQuery = function(hash){
		//
		var encodeSegment = function(segment){
			if( (segment||"")==""){
				return segment;
			}
			var param = segment.split("=");
			var len = param.length;
			var result = [];
			for(var i=0 ; i<len ; i++){
				result.push(encodeURIComponent(param[i]));
			}
			return result.join("=");
		};
		//
		if( (hash||"")=="" ){
			return hash;
		}
		var param = hash.split("&");
		var len = param.length;
		var result = [];
		for(var i=0 ; i<len ; i++){
			result.push(encodeSegment(param[i]));
		}
		return result.join("&");
	};
	// hash
	var sCurHash = top.document.location.hash||"";
	var sNewHash = "";
	if (sCurHash != "") {
		sNewHash = "#" + encodeQuery(sCurHash.substring(1));
	}
	// query string
	var sCurQuery = "";
	var sNewQuery = "";
	if (sCurQuery != ""){
		sNewQuery = "?" + encodeQuery(sCurQuery);
	}
	// URI
	var sURI = "\/jsp\/smartpass\/tchinese\/stress_test.jsp";
	if ( (sURI.indexOf("/english/") > -1) || (sURI.indexOf("/en/") > -1) ) {
		sURI = sURI.replace("/english/", "/tchinese/").replace("/en/", "/tc/");
	} else {
		sURI = sURI.replace("/tchinese/", "/english/").replace("/tc/", "/en/");
	}
	var sTopPath = sURI + sNewQuery + sNewHash;
	if(sURI.indexOf("/preorder_form") > -1){
		sTopPath = sURI;
	}
	sTopPath = sTopPath.replace("#cloud_storage", "");
	//console.log(sTopPath);
	top.location.href = sTopPath;
}
